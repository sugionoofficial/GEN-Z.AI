/* ===================================================== ACCOUNT ACTIVITY ===================================================== */ const accountActivityState = { profiles: [], history: [], selectedUserId: null, refreshTimer: null }; function normalizeActivityStatus(status) { const value = String(status || "pending") .trim() .toLowerCase(); if (value === "completed") { return "success"; } if (value === "queued") { return "pending"; } return value || "pending"; } function isRunningGeneration(item) { const status = normalizeActivityStatus( item?.status ); return ( status === "processing" || status === "pending" ); } function formatActivityDate(value) { if (!value) { return "—"; } const date = new Date(value); if ( Number.isNaN( date.getTime() ) ) { return String(value); } return date.toLocaleString( "id-ID", { dateStyle: "short", timeStyle: "short" } ); } function formatActivityCredits(value) { return Number( value || 0 ).toLocaleString("id-ID"); } async function loadAccountActivity() { const grid = document.getElementById( "accountActivityGrid" ); if (!grid) { return; } try { const profileResult = await supabaseClient .from("profiles") .select( "id,name,email,role,credits,status" ) .order( "email", { ascending: true } ); if ( profileResult.error ) { throw profileResult.error; } const historyResult = await supabaseClient .from("generation_history") .select("*") .order( "created_at", { ascending: false } ); if ( historyResult.error ) { throw historyResult.error; } accountActivityState.profiles = Array.isArray( profileResult.data ) ? profileResult.data : []; accountActivityState.history = Array.isArray( historyResult.data ) ? historyResult.data : []; renderAccountActivity(); } catch (error) { console.error( "[GEN-Z.AI] Account Activity:", error ); grid.innerHTML = `

Belum ada akun yang tersedia.
`; return; } grid.innerHTML = profiles.map( profile => { const userId = String( profile.id || "" ); const userHistory = history.filter( item => String( item.user_id || "" ) === userId ); const activeCount = userHistory.filter( isRunningGeneration ).length; const active = activeCount > 0; return `

${ profile.name || "Tanpa nama" }

`; } else { list.innerHTML = history .map( renderActivityItem ) .join(""); bindCancelButtons(); } } if (modal) { modal.classList.add( "open" ); modal.setAttribute( "aria-hidden", "false" ); } } function renderActivityItem( item ) { const status = normalizeActivityStatus( item.status ); const active = isRunningGeneration( item ); return `
